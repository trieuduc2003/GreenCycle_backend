GreenCycle_BackEnd/
│
├── GreenCycle_BackEnd.sln              # File Solution tổng quản lý toàn bộ dự án Backend
│
├── 1. Core/
│   ├── GreenCycle.Domain/              # TẦNG THỰC THỂ (Tuyệt đối không tham chiếu project nào)
│   │   ├── Entities/                   # Chứa các bảng Database dưới dạng Class C#
│   │   │   ├── User.cs                 # Thông tin tài khoản (Sinh viên, Tài xế, Admin)
│   │   │   ├── Order.cs                # Đơn hàng thu gom rác (Vị trí GPS, khối lượng, trạng thái)
│   │   │   ├── WasteCategory.cs        # Danh mục rác (Nhựa, giấy, kim loại...) và giá tiền
│   │   │   ├── Reward.cs               # Quà tặng/Voucher hệ thống đối tác
│   │   │   └── Transaction.cs          # Lịch sử tích điểm hoặc đổi quà
│   │   ├── Enums/                      # Định nghĩa các hằng số trạng thái cố định
│   │   │   ├── OrderStatus.cs          # Chờ duyệt, Tài xế nhận đơn, Đang cân, Hoàn thành, Hủy
│   │   │   └── UserRole.cs             # Student, Driver, Admin
│   │   └── Common/                     # Các class bổ trợ dùng chung cho tầng Domain
│   │
│   └── GreenCycle.Application/         # TẦNG LOGIC NGHIỆP VỤ (Chỉ tham chiếu đến Domain)
│       ├── Interfaces/                 # Định nghĩa các bản hợp đồng giao tiếp (Interface)
│       │   ├── IApplicationDbContext.cs
│       │   ├── IMinioStorageService.cs # Interface cho chức năng lưu ảnh
│       │   ├── IJwtTokenService.cs     # Interface cho chức năng tạo token bảo mật
│       │   └── ISignalRService.cs      # Interface điều phối thông báo realtime
│       ├── DTOs/                       # Gói dữ liệu thô (Data Transfer Object) gửi ra bên ngoài
│       │   ├── Auth/ (LoginDto.cs, RegisterDto.cs, TokenDto.cs)
│       │   ├── Orders/ (CreateOrderDto.cs, OrderDto.cs, ConfirmWeightDto.cs)
│       │   └── Rewards/ (RewardDto.cs, RedeemVoucherDto.cs)
│       ├── Services/                   # Nơi viết code xử lý thuật toán cốt lõi của app
│       │   ├── OrderService.cs         # Tính toán điểm thưởng, xử lý luồng đơn hàng
│       │   ├── UserService.cs          # Quản lý số dư điểm ví, hồ sơ cá nhân
│       │   └── RewardService.cs        # Xử lý logic trừ điểm đổi mã Voucher
│       └── Exceptions/                 # Tự định nghĩa các lỗi nghiệp vụ hệ thống
│
├── 2. Infrastructure/
│   └── GreenCycle.Infrastructure/      # TẦNG KỸ THUẬT NGOẠI VI (Tham chiếu Application & Domain)
│       ├── Persistence/                # Kết nối Database và cấu hình Entity Framework Core
│       │   ├── ApplicationDbContext.cs # Class trung gian trực tiếp điều khiển dữ liệu SQL Server
│       │   ├── Migrations/             # Thư mục tự sinh lưu lịch sử đồng bộ database của EF Core
│       │   └── SeedData/               # Khởi tạo dữ liệu mẫu chạy lần đầu (Nạp sẵn giá các loại rác)
│       ├── Services/                   # Triển khai chi tiết code (Implement) từ Interface ở tầng App
│       │   ├── MinioStorageService.cs  # Viết code đẩy file ảnh lên MinIO và trả về link URL tĩnh
│       │   └── JwtTokenService.cs      # Viết code tạo chuỗi mã hóa Token JWT để gửi về Mobile App
│       └── Realtime/                   # Thiết lập kênh truyền thời gian thực
│           ├── NotificationHub.cs      # Tạo trạm kết nối ngầm (Hub SignalR)
│           └── SignalRService.cs       # Code bắn tín hiệu tọa độ xe và cập nhật đơn hàng tức thì
│
└── 3. Presentation/
    └── GreenCycle.WebApi/              # TẦNG CỬA NGÕ API (Tham chiếu Infrastructure & Application)
        ├── Controllers/                # Tiếp nhận các request HTTP từ ứng dụng di động (.NET MAUI)
        │   ├── AuthController.cs       # Endpoint Đăng ký, Đăng nhập
        │   ├── OrdersController.cs     # Endpoint Đặt lịch gom rác, Tài xế nhận đơn, Cân rác đối soát
        │   ├── WasteCategoriesController.cs # Endpoint lấy bảng danh mục và bảng giá rác
        │   └── RewardsController.cs    # Endpoint thực hiện đổi điểm lấy Voucher
        ├── Middlewares/                # Bộ lọc bắt lỗi hệ thống tự động để Server không bị sập
        │   └── ExceptionHandlingMiddleware.cs
        ├── appsettings.json            # Lưu Connection String SQL Server, cấu hình tài khoản MinIO, JWT
        └── Program.cs                  # File chạy chính: Đăng ký Dependency Injection và kích hoạt Server
